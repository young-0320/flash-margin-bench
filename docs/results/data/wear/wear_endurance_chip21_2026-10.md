L=build/inverse/loco_chip21 && mkdir -p $L/with21 && cp docs/results/data/wear_curves/wear_curves_chip0{1,2,3,4,6,7,9}_2026-*.csv docs/results/data/wear_curves/wear_curves_chip{15,18}_2026-10.csv docs/results/data/wear_curves/wear_curves_chip21_nocal_2026-10.csv $L/with21/ && mv $L/with21/wear_curves_chip21_nocal_2026-10.csv $L/with21/wear_curves_chip21_2026-10.csv
